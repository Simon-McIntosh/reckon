Read-only census of the last seven days of the reckon ledger: for every review whose ledger row falls in the window, its repair rounds and where each finding was answered with a commit or declined at a reason.

Method

  • Population. The committed ledger rows under docs/state/reckon/runs/ whose node name begins review-of- and whose completed_at (fallback dispatched_at) falls inside the window. A review is counted once per ledger row.
  • Findings. Read from the stored review record the row joins (by the script set review_run_id, falling back to the reviewed node name). Finding ids are minted by reckon.crew.repair.finding_id — the census imports the code's own composer rather than re-deriving the ids.
  • A repair round is identified as repair_round_id (the reviewed run and the head the review read). The census counts evidence of a round from three durable records: the reviewed run's live pointer (repair_rounds/repair_dispatch, written by the reflex), a resume-*-advice.txt in the reviewed run's own directory whose text names that round's blocking finding ids, and a ledger row whose node id equals the composed repair node id repair-of-<node>-<head12>.
  • An answer is read from the reviewed run's manifest: a line acted <finding-id>: <commit> or declined <finding-id>: <reason>. A finding with no such line and a declared follow-on severity is recorded as declined (the reviewer's own severity is the decline); a finding with no line at all is recorded as unanswered.

#!/usr/bin/env python3
"""Weekly census of repair rounds per finding-bearing review, read from the ledger.

Reads, for one project (default ``reckon``):

* the committed ledger rows          ``<repo>/docs/state/reckon/runs/*.json``
* the live run pointers              ``~/.config/reckon/crew/live/*.json``
* the stored review records          ``~/.config/reckon/crew/reviews/<project>/*.json``
* the run directories                ``~/.config/reckon/crew/runs/<run-id>/``
  (repair resume advice and the reviewed run's manifest)

and reports, over the last seven days: how many reviews were dispatched,
how many carried findings, the repair rounds of each finding-bearing review,
and, per finding, whether it was answered with a commit or declined with a
reason. Read-only: nothing is written.

Run from the repository root with the project's environment:

    cd <checkout>
    PYTHONPATH="$PWD" .venv/bin/python <this-script> [--end ISO8601]
"""

from __future__ import annotations

import argparse
import json
import re
import sys
from datetime import UTC, datetime, timedelta
from pathlib import Path

from reckon.crew import repair as repair_module

# Marker the reflex writes at the head of a composed repair resume. It names
# the blocking findings the round carries, so an advice file that repeats the
# ids was written for that round and counts as an attempt of it.
_REPAIR_ADVICE = re.compile(r"An independent review of this run found (\d+) blocking finding")

# Answer lines the repair's own contract puts in the reviewed run's manifest:
# ``acted <finding-id>: <commit>`` or ``declined <finding-id>: <reason>``.
_ACTED = re.compile(r"^\s*acted\s+(f[0-9a-f]{8,}):\s*(\S.*)$", re.MULTILINE)
_DECLINED = re.compile(r"^\s*declined\s+(f[0-9a-f]{8,}):\s*(\S.*)$", re.MULTILINE)

ANSWER_LINE = re.compile(r"^\s*(acted|declined)\s+(f[0-9a-f]{8,})\s*:\s*(.*)$", re.MULTILINE)


def parse_utc(text: str | None) -> datetime | None:
    if not text:
        return None
    try:
        moment = datetime.fromisoformat(str(text).replace("Z", "+00:00"))
    except ValueError:
        return None
    return moment if moment.tzinfo else moment.replace(tzinfo=UTC)


def deadline_of(row: dict) -> datetime | None:
    return parse_utc(row.get("completed_at")) or parse_utc(row.get("dispatched_at"))


def load_rows(root: Path) -> dict[str, dict]:
    result: dict[str, dict] = {}
    directory = Path(root)
    if not directory.is_dir():
        return result
    for path in directory.glob("*.json"):
        try:
            row = json.loads(path.read_text())
        except (OSError, ValueError):
            continue
        run_id = str(row.get("run_id") or path.stem)
        row["_path"] = str(path)
        result[run_id] = row
    return result


def load_reviews(root: Path) -> list[dict]:
    records: list[dict] = []
    directory = Path(root)
    if not directory.is_dir():
        return records
    for path in directory.glob("*.json"):
        try:
            record = json.loads(path.read_text())
        except (OSError, ValueError):
            continue
        record["_path"] = str(path)
        records.append(record)
    return records


def review_moment(record: dict, path: Path) -> datetime:
    stated = parse_utc(record.get("timestamp"))
    if stated:
        return stated
    return datetime.fromtimestamp(path.stat().st_mtime, tz=UTC)


def round_evidence(review: dict, reviewed_run_id: str, run_dirs: Path, ledger: dict[str, dict]) -> dict:
    """Rounds observed for one review, from every record the crew keeps."""
    round_id = repair_module.repair_round_id(review, reviewed_run_id=reviewed_run_id)
    head = round_id.split("@")[-1] if "@" in round_id else ""
    evidence: dict = {"round_id": round_id, "rounds": 0, "sources": [], "attempts": 0, "records": []}

    pointer_path = Path(run_dirs).parent / "live" / f"{reviewed_run_id}.json"
    pointer: dict = {}
    if pointer_path.exists():
        try:
            pointer = json.loads(pointer_path.read_text())
        except (OSError, ValueError):
            pointer = {}
    opened = pointer.get("repair_rounds") if isinstance(pointer.get("repair_rounds"), dict) else None
    dispatch = pointer.get("repair_dispatch") if isinstance(pointer.get("repair_dispatch"), dict) else None
    if opened:
        evidence["rounds"] = int(opened.get("count") or 0)
        evidence["attempts"] = int(opened.get("attempts") or 0)
        evidence["sources"].append("live-pointer repair_rounds")
        if str(opened.get("round_id") or "") != round_id:
            evidence["sources"].append("live-pointer carries a DIFFERENT round")
    if dispatch:
        evidence["records"].append(
            {"at": dispatch.get("at"), "status": dispatch.get("status"), "reason": dispatch.get("reason")}
        )
        evidence["sources"].append("live-pointer repair_dispatch")

    # Advice files: a reflex resume writes the composed repair into the reviewed
    # run's own directory. The composed advice names the round's own blocking
    # finding ids, so an advice file counts for this round only when every id
    # this review's round carries appears in it — two reviews of one run at two
    # heads write two advices, and each belongs to its own round.
    blocking_ids = [
        repair_module.finding_id(finding)
        for finding in repair_module.review_findings(review)
        if repair_module._finding_kind(finding) == "blocking"
    ]
    advice_dir = Path(run_dirs) / reviewed_run_id
    advice_rounds: set[str] = set()
    if advice_dir.is_dir() and blocking_ids:
        for advice in sorted(advice_dir.glob("resume-*-advice.txt")):
            try:
                text = advice.read_text(errors="replace")
            except OSError:
                continue
            match = _REPAIR_ADVICE.search(text)
            if match and all(fid in text for fid in blocking_ids):
                advice_rounds.add(advice.name)
                evidence["sources"].append(f"advice:{advice.name}")
    if advice_rounds and evidence["rounds"] == 0:
        evidence["rounds"] = 1
        evidence["sources"].append("advice-only round")

    # A dispatch of the composed node leaves its own ledger row.
    source_node = ""
    reviewed_row = ledger.get(reviewed_run_id) or {}
    if reviewed_row:
        source_node = str(reviewed_row.get("node") or "")
    if head and source_node:
        node_id = repair_module.repair_node_id(review, reviewed_run_id=reviewed_run_id, source_node=source_node)
        for run_id, row in ledger.items():
            if str(row.get("node") or "") == node_id:
                evidence["records"].append(
                    {"run_id": run_id, "node": node_id, "at": row.get("completed_at") or row.get("dispatched_at")}
                )
                evidence["sources"].append(f"ledger row {run_id}")
                evidence["rounds"] = max(evidence["rounds"], 1)
    return evidence


def manifest_of(reviewed_run_id: str, folder: Path, ledger: dict[str, dict]) -> Path | None:
    row = ledger.get(reviewed_run_id)
    if row and row.get("manifest_path"):
        candidate = Path(str(row["manifest_path"]))
        if candidate.exists():
            return candidate
    candidate = Path(folder) / reviewed_run_id / "manifest.md"
    return candidate if candidate.exists() else None


def answers_from(text: str) -> dict[str, dict]:
    out: dict[str, dict] = {}
    for kind, fid, rest in ANSWER_LINE.findall(text):
        out.setdefault(fid, {"action": kind, "rest": rest.strip()})
    return out


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--repo", default=".")
    parser.add_argument("--project", default="reckon")
    parser.add_argument("--crew-home", default=str(Path.home() / ".config" / "reckon" / "crew"))
    parser.add_argument("--end", default=None, help="window end (UTC ISO8601); default now")
    parser.add_argument("--days", type=int, default=7)
    parser.add_argument("--json", default=None, help="write the census as JSON here")
    args = parser.parse_args()

    end = parse_utc(args.end) or datetime.now(tz=UTC)
    start = end - timedelta(days=args.days)
    repo = Path(args.repo).resolve()
    crew_home = Path(args.crew_home)
    ledger = load_rows(repo / "docs" / "state" / args.project / "runs")
    live = load_rows(crew_home / "live")
    reviews = load_reviews(crew_home / "reviews" / args.project)

    by_review_run = {str(r.get("review_run_id") or ""): r for r in reviews if r.get("review_run_id")}

    # The population: review ledger rows dispatched or completed inside the window.
    review_rows = {
        run_id: row
        for run_id, row in ledger.items()
        if str(row.get("node") or "").startswith("review-of-")
        and (deadline_of(row) is not None and start <= deadline_of(row) <= end)
    }

    counted: list[dict] = []
    for run_id, row in sorted(review_rows.items(), key=lambda kv: str(deadline_of(kv[1]))):
        record = by_review_run.get(run_id)
        if record is None:
            # Fall back to the reviewed node the run's own name declares: a
            # record stored by an older writer carries no review_run_id, but its
            # reviewed_run_id still resolves to the ledger row named after it.
            source_node = str(row.get("node") or "").removeprefix("review-of-")
            for candidate_run, candidate_row in ledger.items():
                if str(candidate_row.get("node") or "") == source_node:
                    found = next(
                        (r for r in reviews if str(r.get("reviewed_run_id") or "") == candidate_run),
                        None,
                    )
                    if found is not None:
                        record = found
                    break
        findings = repair_module.review_findings(record) if record else []
        reviewed_run_id = str((record or {}).get("reviewed_run_id") or "")
        entry: dict = {
            "review_run_id": run_id,
            "when": str(deadline_of(row)),
            "node": row.get("node"),
            "score": (record or {}).get("total"),
            "reviewed_run_id": reviewed_run_id,
            "findings": [],
            "rounds": 0,
            "round_sources": [],
        }
        if findings:
            evidence = round_evidence(record, reviewed_run_id, crew_home / "runs", {**ledger, **live})
            entry["rounds"] = evidence["rounds"]
            entry["round_sources"] = evidence["sources"]
            entry["round_records"] = evidence["records"]
            entry["round_id"] = evidence["round_id"]
            folder = manifest_dir = crew_home / "runs"
            manifest_path = manifest_of(reviewed_run_id, folder, {**ledger, **live})
            answers: dict[str, dict] = {}
            if manifest_path is not None:
                try:
                    answers = answers_from(manifest_path.read_text(errors="replace"))
                except OSError:
                    answers = {}
            for finding in findings:
                fid = repair_module.finding_id(finding)
                kind = repair_module._finding_kind(finding)
                answer = answers.get(fid)
                if answer:
                    disposition = f"{answer['action']}: {answer['rest']}"
                elif kind == "follow-on":
                    disposition = "declined: follow-on, recorded on the review and not repaired"
                else:
                    disposition = "unanswered"
                entry["findings"].append(
                    {
                        "id": fid,
                        "kind": kind,
                        "severity": finding.get("severity"),
                        "file": finding.get("file"),
                        "line": finding.get("line"),
                        "disposition": disposition,
                    }
                )
        counted.append(entry)

    finding_bearing = [e for e in counted if e["findings"]]
    payload = {
        "project": args.project,
        "window_start": start.isoformat(),
        "window_end": end.isoformat(),
        "review_rows_in_window": len(counted),
        "review_rows_without_a_stored_record": sum(1 for e in counted if not e["reviewed_run_id"]),
        "finding_bearing_reviews": len(finding_bearing),
        "findings_total": sum(len(e["findings"]) for e in finding_bearing),
        "reviews_with_more_than_one_round": [
            e["review_run_id"] for e in finding_bearing if e["rounds"] > 1
        ],
        "reviews_by_rounds": {
            str(n): sum(1 for e in finding_bearing if e["rounds"] == n)
            for n in sorted({e["rounds"] for e in finding_bearing})
        },
        "dispositions": {},
        "reviews": counted,
    }
    tally: dict[str, int] = {}
    for entry in finding_bearing:
        for finding in entry["findings"]:
            head = finding["disposition"].split(":", 1)[0]
            tally[head] = tally.get(head, 0) + 1
    payload["dispositions"] = tally

    if args.json:
        Path(args.json).write_text(json.dumps(payload, indent=2))

    print(f"window {payload['window_start']} .. {payload['window_end']}")
    print(f"reviews counted (ledger rows): {payload['review_rows_in_window']}")
    print(f"  with no stored record joined: {payload['review_rows_without_a_stored_record']}")
    print(f"finding-bearing reviews: {payload['finding_bearing_reviews']}")
    print(f"findings: {payload['findings_total']}  dispositions: {payload['dispositions']}")
    print("rounds distribution:", payload["reviews_by_rounds"])
    print("reviews with >1 round:", payload["reviews_with_more_than_one_round"] or "none")
    for entry in finding_bearing:
        print(
            f"  {entry['review_run_id']} score={entry['score']} rounds={entry['rounds']} "
            f"findings={len(entry['findings'])} sources={entry['round_sources']}"
        )
        for finding in entry["findings"]:
            print(f"      {finding['id']} [{finding['kind']}] {finding['file']}:{finding['line']} -> {finding['disposition']}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
