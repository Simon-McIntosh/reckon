"""Census reckon MCP call outcomes and CLI/HTML fallbacks in session transcripts."""
import json, re, sys
from collections import Counter
calls, outcomes, fallbacks = Counter(), Counter(), Counter()
per_tool_bad = Counter()
ERR = re.compile(r'"error"\s*:\s*"([a-z_\-]+)"|\\"error\\"\s*:\s*\\"([a-z_\-]+)\\"')
FALLBACK = {
    "cli roadmap/status": re.compile(r"reckon (roadmap|status|audit)\b"),
    "cli crew read": re.compile(r"reckon crew (list|live|runs|show|observe|scopes|obligations|status)\b"),
    "parse plan html": re.compile(r"(python|grep|sed|awk|rg)[^\n]*docs/(plans|sprints)/[^\s]*\.html"),
    "read crew json/pointers": re.compile(r"(crew\.json|crew/live/|crew/runs/[^ ]*/manifest)"),
}
for path in sys.argv[1:]:
    names = {}
    with open(path, errors="replace") as fh:
        for line in fh:
            if '"tool_use"' not in line and '"tool_result"' not in line:
                continue
            try:
                rec = json.loads(line)
            except Exception:
                continue
            content = (rec.get("message") or {}).get("content")
            if not isinstance(content, list):
                continue
            for block in content:
                if not isinstance(block, dict):
                    continue
                if block.get("type") == "tool_use":
                    name = block.get("name", "")
                    names[block.get("id")] = name
                    if name.startswith("mcp__reckon__"):
                        view = (block.get("input") or {}).get("view") or (block.get("input") or {}).get("action") or "-"
                        calls[(name[13:], view)] += 1
                    elif name == "Bash":
                        cmd = (block.get("input") or {}).get("command", "")
                        for label, rx in FALLBACK.items():
                            if rx.search(cmd):
                                fallbacks[label] += 1
                elif block.get("type") == "tool_result":
                    name = names.get(block.get("tool_use_id"), "")
                    if not name.startswith("mcp__reckon__"):
                        continue
                    text = json.dumps(block.get("content"))[:4000]
                    m = ERR.search(text)
                    kind = (m.group(1) or m.group(2)) if m else ("is_error" if block.get("is_error") else "ok")
                    outcomes[kind] += 1
                    if kind != "ok":
                        per_tool_bad[(name[13:], kind)] += 1
total = sum(outcomes.values())
print("reckon MCP calls:", sum(calls.values()), " results:", total)
for (tool, view), n in calls.most_common(25):
    print(f"  {n:5d}  {tool}:{view}")
print("outcomes:")
for k, n in outcomes.most_common():
    print(f"  {n:5d}  {k}  ({100*n/max(total,1):.1f}%)")
print("non-ok by tool:")
for (tool, kind), n in per_tool_bad.most_common(15):
    print(f"  {n:5d}  {tool}  {kind}")
print("fallback Bash commands:")
for k, n in fallbacks.most_common():
    print(f"  {n:5d}  {k}")

§10 — a moved checkout serves every call on current code

Run moved-checkout-serves-every-call, landed at 564b134aa after four repair rounds.
