"""Prototype: cross-plan section-level critical path for one project."""
import html, json, re, sys
from pathlib import Path

plans_dir = Path(sys.argv[1])
meta_re = lambda name: re.compile(r'<meta name="%s" content="([^"]*)"' % name)
sec_re = re.compile(r'<section data-reckon="section"([^>]*)>')
attr = lambda s, a: (re.search(r'%s="([^"]*)"' % a, s) or [None, None])[1]
title_re = re.compile(r'<h2[^>]*>(.*?)</h2>', re.S)

nodes, edges, plan_meta = {}, [], {}
for f in sorted(plans_dir.glob("*.html")):
    text = f.read_text()
    slug = f.stem
    def m(name):
        hit = meta_re(name).search(text)
        return html.unescape(hit.group(1)) if hit else None
    status = m("plan-status")
    plan_meta[slug] = {"status": status, "sprint": m("plan-sprint")}
    decl = json.loads(m("plan-section-declarations") or "{}")
    for hit in sec_re.finditer(text):
        a = hit.group(1); sid = attr(a, "data-id")
        h2 = re.search(r'<h2 id="%s"[^>]*>(.*?)</h2>' % sid, text, re.S)
        name = html.unescape(re.sub(r"<[^>]+>", "", h2.group(1)).strip()) if h2 else ""
        done = decl.get(sid) == "done" or attr(a, "data-status") == "done" or status in ("done", "shipped", "archived", "superseded")
        nodes[f"{slug}#{sid}"] = {"effort": float(attr(a, "data-effort-hours") or 0), "done": done,
                                  "decl": decl.get(sid), "title": name[:70], "sprint": plan_meta[slug]["sprint"]}
    deps = json.loads(m("plan-section-depends-on") or "{}")
    for src, targets in (deps.items() if isinstance(deps, dict) else []):
        for t in targets if isinstance(targets, list) else [targets]:
            t = t if "#" in t else f"{slug}#{t}"
            edges.append((t, f"{slug}#{src}"))

open_nodes = {k for k, v in nodes.items() if not v["done"]}
succ, pred = {}, {}
for a, b in edges:
    if b in open_nodes and (a in open_nodes or a not in nodes):
        succ.setdefault(a, set()).add(b); pred.setdefault(b, set()).add(a)
memo = {}
def longest(n):  # longest remaining chain starting at n (inclusive)
    if n in memo: return memo[n]
    best = (0.0, [])
    for s in succ.get(n, ()):
        cand = longest(s)
        if cand[0] > best[0]: best = cand
    w = nodes.get(n, {}).get("effort", 0) if n in open_nodes else 0
    memo[n] = (w + best[0], [n] + best[1]); return memo[n]
def downstream(n, seen=None):
    seen = set() if seen is None else seen
    for s in succ.get(n, ()):
        if s not in seen: seen.add(s); downstream(s, seen)
    return seen
roots = sorted(open_nodes, key=lambda n: -longest(n)[0])
print("open sections:", len(open_nodes), " cross-plan edges:", sum(1 for a,b in edges if a.split('#')[0]!=b.split('#')[0]))
print("\nCRITICAL PATH (effort-weighted, sections):")
L, path = longest(roots[0])
for n in path: print(f"  {n:48s} {nodes[n]['effort']:5.1f}h  {nodes[n]['title']}")
print(f"  total {L:.1f} worker-hours")
print("\nTOP OPEN SECTIONS by downstream reach (sections they transitively unblock):")
rank = sorted(open_nodes, key=lambda n: (-len(downstream(n)), -longest(n)[0]))
for n in rank:
    ready = all(p not in open_nodes for p in pred.get(n, ()))
    print(f"  {n:48s} unblocks {len(downstream(n)):2d}  chain {longest(n)[0]:5.1f}h  {'READY ' if ready else 'waits '} {nodes[n]['title']}")
