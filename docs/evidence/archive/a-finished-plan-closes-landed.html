HTML census script

"""Census: plans whose sections are all done but which are not closed, and what holds them."""
import html, json, re, sys
from collections import Counter
from pathlib import Path

TERMINAL = {"done", "shipped", "archived", "superseded", "abandoned"}
mounts = json.load(open(sys.argv[1]))
meta = lambda t, n: (lambda m: html.unescape(m.group(1)) if m else "")(re.search(r'<meta name="%s" content="([^"]*)"' % n, t))
decl_values, rows, statuses = Counter(), [], Counter()
for project, docs in sorted(mounts.items()):
    docs = docs if isinstance(docs, str) else docs.get("docs", "")
    for f in sorted(Path(docs, "plans").glob("*.html")):
        t = f.read_text(errors="replace")
        if meta(t, "reckon-type") not in ("", "plan"):
            continue
        status = meta(t, "plan-status")
        try:
            decl = json.loads(meta(t, "plan-section-declarations") or "{}")
        except ValueError:
            decl = {}
        decl_values.update(decl.values())
        if not decl or status in TERMINAL:
            continue
        open_secs = {k: v for k, v in decl.items() if v != "done"}
        if open_secs and set(open_secs.values()) - {"deferred"}:
            continue  # real open work remains
        gates = re.findall(r'<div class="r-gate" data-id="([^"]*)" data-section="([^"]*)"[^>]*data-status="([^"]*)" data-verdict="([^"]*)"', t)
        held_gates = [f"{g}({s}:{st}/{v or '-'})" for g, s, st, v in gates if st != "closed" or v == "failed"]
        fus = re.findall(r'<article class="r-fu" data-id="([^"]*)"([^>]*)>', t)
        open_fus = [i for i, a in fus if 'data-resolved-at=""' in a or "data-resolved-at" not in a]
        deps = [d for d in meta(t, "plan-depends-on").split(",") if d.strip()]
        statuses[status] += 1
        rows.append((project, f.stem, status, len(decl), sorted(open_secs.items()), held_gates, open_fus, deps, meta(t, "plan-modified")))
print("declaration values across all plans:", dict(decl_values))
